        function filterSociologists() {

            const query = searchInput.value.trim().toLowerCase();

            const fieldQuery = searchField.value;

            const start = parseInt(yearStart.value) || 0;

            const end = parseInt(yearEnd.value) || 9999;


            const filtered = sociologists.filter(p => {

                // 全文匹配：姓名、英文名、案例、著作、简介

                const textMatch = !query || 

                    p.name.toLowerCase().includes(query) || 

                    p.nameEn.toLowerCase().includes(query) ||

                    p.cases.some(c => c.toLowerCase().includes(query)) ||

                    p.代表作.some(w => w.toLowerCase().includes(query)) ||

                    p.description.toLowerCase().includes(query);

                

                // 流派匹配

                const fieldMatch = !fieldQuery || p.field === fieldQuery;

                

                // 年份匹配（活跃期有重叠即命中）

                const yearMatch = p.activeEnd >= start && p.activeStart <= end;


                return textMatch && fieldMatch && yearMatch;

            });


            renderCards(filtered);

        }


        // 绑定事件

        searchInput.addEventListener('input', filterSociologists);

        searchField.addEventListener('change', filterSociologists);

        yearStart.addEventListener('input', filterSociologists);

        yearEnd.addEventListener('input', filterSociologists);


        resetBtn.addEventListener('click', () => {

            searchInput.value = '';

            searchField.value = '';

            yearStart.value = '';

            yearEnd.value = '';

            filterSociologists();

        });


        // 初始化渲染

        filterSociologists();
